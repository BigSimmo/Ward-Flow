from pathlib import Path
import re
text = Path(r'D:\Worktrees\Database\ward-lead\_wt_list.txt').read_text(encoding='utf-8', errors='replace')
entries = []
cur = {}
for line in text.splitlines():
    if line.startswith('worktree '):
        if cur: entries.append(cur)
        cur = {'path': line[9:].strip()}
    elif line.startswith('HEAD '):
        cur['head'] = line[5:].strip()[:12]
    elif line.startswith('branch '):
        cur['branch'] = line[7:].strip().replace('refs/heads/','')
    elif line.startswith('detached'):
        cur['branch'] = '(detached)'
    elif line.startswith('locked'):
        cur['locked'] = True
if cur: entries.append(cur)

# antigravity / gemini with ward-ish
anti = [e for e in entries if 'antigravity' in e.get('path','').lower() or 'gemini' in e.get('path','').lower()]
print(f'ANTIGRAVITY_OR_GEMINI_WT {len(anti)}')
wardish = re.compile(r'ward|flow|elevation|chrome', re.I)
for e in anti:
    blob = f"{e.get('path','')} {e.get('branch','')}"
    mark = ' <<WARDISH' if wardish.search(blob) else ''
    print(f"{e.get('branch','?'):55}{mark}")
    print(f"  {e.get('path')}")

print('--- primary ---')
# tip of ward-lead
import subprocess
r=subprocess.run(['git','-C',r'D:\Worktrees\Database\ward-lead','log','-1','--oneline'], capture_output=True, text=True)
print('ward-lead tip', r.stdout.strip())
